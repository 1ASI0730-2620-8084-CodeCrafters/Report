workspace "BottleTrack" "Beverage distribution management system" {

    !identifiers hierarchical

    model {

        administrator = person "Administrator" "Manages users, roles and distribution operations."

        fleetSupervisor = person "Fleet Supervisor" "Plans operations, assigns vehicles and drivers, monitors the fleet and manages incidents."

        driver = person "Driver" "Executes deliveries, confirms delivery status and reports incidents."


        bottleTrack = softwareSystem "BottleTrack" "Beverage distribution management system." {

            landingPage = container "Landing Page" "Presents BottleTrack, its value proposition and information about the startup." "HTML5 / CSS3 / JavaScript"

            webApplication = container "Web Application" "Provides the user interface for managing and monitoring distribution operations." "Vue.js / PrimeVue"

            backendApi = container "backend API" "Provides the business logic and API endpoints for BottleTrack." "ASP.NET Core / C# / Entity Framework Core"

            database = container "Database" "Stores users, vehicles, drivers, distribution operations, deliveries and incidents." "PostgreSQL"
        }


        administrator -> bottleTrack.landingPage "Visits the landing page"

        fleetSupervisor -> bottleTrack.landingPage "Visits the landing page"

        driver -> bottleTrack.landingPage "Visits the landing page"


        administrator -> bottleTrack.webApplication "Manages users and operations"

        fleetSupervisor -> bottleTrack.webApplication "Plans, assigns and monitors distribution operations"

        driver -> bottleTrack.webApplication "Views assigned operations, confirms deliveries and reports incidents"


        bottleTrack.webApplication -> bottleTrack.backendApi "Makes API requests" "HTTPS/JSON"

        bottleTrack.backendApi -> bottleTrack.database "Reads from and writes to" "Entity Framework Core"
    }


    views {

        container bottleTrack "ContainerDiagram" {

            include administrator
            include fleetSupervisor
            include driver

            include bottleTrack.landingPage
            include bottleTrack.webApplication
            include bottleTrack.backendApi
            include bottleTrack.database

            autolayout lr

            title "Container View - BottleTrack"
        }


        styles {

            element "Person" {
                shape person
                background #08427b
                color #ffffff
            }

            element "Software System" {
                background #1168bd
                color #ffffff
            }

            element "Container" {
                background #438dd5
                color #ffffff
            }

            relationship "Relationship" {
                thickness 2
            }
        }
    }
}

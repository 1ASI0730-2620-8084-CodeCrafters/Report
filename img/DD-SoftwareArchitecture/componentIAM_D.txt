workspace "BottleTrack - Identity and Access Management" "Component diagram" {

    !identifiers hierarchical

    model {
        administrator = person "Distributor Administrator" "Manages the company, its users, its fleet and reviews operation indicators."
        fleetSupervisor = person "Fleet Supervisor" "Plans transport operations, records delivery results and follows up incidents."
        deliveryPointOwner = person "Delivery Point Owner" "Owner of a convenience store or minimarket that tracks its orders and reports reception problems."
        driver = person "Driver" "Drives the vehicle and reports delivery results to the fleet supervisor. Does not use the application."
        iotDevice = softwareSystem "Vehicle IoT Device" "Sends location, cargo temperature and impact readings during the trip." "External"
        mapsService = softwareSystem "Maps and Geolocation Service" "Third party service that geocodes delivery addresses and draws routes." "External"

        bottleTrack = softwareSystem "BottleTrack" "Manages and tracks bottled beverage transport operations with IoT monitoring and incident reporting." {
            landingPage = container "Landing Page" "Presents the business model and redirects each segment to the web application." "HTML5, CSS3, JavaScript - GitHub Pages"
            webApplication = container "Web Application" "Single page application organized by bounded contexts." "Vue 3, PrimeVue, Pinia, Vue Router, vue-i18n, axios - Vercel"
            backendApi = container "Backend API" "RESTful API organized as a modular monolith, one module per bounded context, each one with a hexagonal architecture." "ASP.NET Core, C#, Entity Framework Core" {
                identityController = component "Authentication and Users Controllers" "Inbound REST adapter. Exposes sign in, users, roles and organization endpoints." "ASP.NET Core Controller"
                identityCommandService = component "Identity Command Service" "Application service. Signs users in, registers users, assigns roles and updates the organization." "C# Application Service"
                identityQueryService = component "Identity Query Service" "Application service. Resolves the profile, users and organization data." "C# Application Service"
                identityDomain = component "Identity Domain Model" "Aggregates User and Organization, value objects EmailAddress and Role." "C# Domain Model"
                identityPort = component "User and Organization Repositories" "Outbound ports declared by the domain." "C# Interfaces" "Port"
                identityAdapter = component "EF Core Identity Repositories" "Outbound persistence adapter that implements the repository ports." "Entity Framework Core"
                identityToken = component "Token Service" "Outbound adapter that issues and validates JSON Web Tokens." "ASP.NET Core Authentication"
            }
            database = container "Database" "Stores the data of every bounded context in its own set of tables." "MySQL" "Database"
        }

        administrator -> bottleTrack.webApplication "Uses" "HTTPS"
        fleetSupervisor -> bottleTrack.webApplication "Uses" "HTTPS"
        deliveryPointOwner -> bottleTrack.webApplication "Uses" "HTTPS"
        bottleTrack.webApplication -> bottleTrack.backendApi.identityController "Makes API requests" "HTTPS / JSON"
        bottleTrack.backendApi.identityController -> bottleTrack.backendApi.identityCommandService "Sends commands"
        bottleTrack.backendApi.identityController -> bottleTrack.backendApi.identityQueryService "Sends queries"
        bottleTrack.backendApi.identityCommandService -> bottleTrack.backendApi.identityDomain "Applies business rules"
        bottleTrack.backendApi.identityCommandService -> bottleTrack.backendApi.identityPort "Persists aggregates through"
        bottleTrack.backendApi.identityQueryService -> bottleTrack.backendApi.identityPort "Reads through"
        bottleTrack.backendApi.identityAdapter -> bottleTrack.backendApi.identityPort "Implements"
        bottleTrack.backendApi.identityAdapter -> bottleTrack.database "Reads from and writes to" "SQL"
        bottleTrack.backendApi.identityCommandService -> bottleTrack.backendApi.identityToken "Issues tokens with"
    }

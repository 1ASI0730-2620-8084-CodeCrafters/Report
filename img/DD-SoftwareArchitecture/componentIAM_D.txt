                userRepository = component "User Repository" "Provides persistence operations for users and roles." "Entity Framework Core"
            }

        bottleTrack.backendApi.identityAccessService -> bottleTrack.backendApi.userRepository "Manages user and role data"

        bottleTrack.backendApi.organizationRepository -> bottleTrack.database "Reads from and writes to" "Entity Framework Core"

        bottleTrack.backendApi.userRepository -> bottleTrack.database "Reads from and writes to" "Entity Framework Core"
